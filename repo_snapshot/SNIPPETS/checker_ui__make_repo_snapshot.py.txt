#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
make_repo_snapshot.py
在项目根目录运行，生成 repo_snapshot/ 目录与 repo_snapshot.zip，包含：
- TREE.txt: 目录树（忽略 .venv/ dist/ build/ .git 等）
- ARCHITECTURE.md: 架构概要（子目录角色、入口、依赖大图概述）
- PY_INDEX.csv: Python 文件索引（大小、哈希、行数）
- SNIPPETS/: 每个关键源码文件的头部与类/函数签名摘要
"""

import argparse, os, sys, hashlib, zipfile, io, ast, textwrap, json
from pathlib import Path

IGNORE_DIRS = {".git", ".venv", "venv", "build", "dist", "__pycache__", ".idea", ".vscode", ".mypy_cache"}
TEXT_EXTS   = {".py", ".ui", ".spec", ".yml", ".yaml", ".json", ".md", ".txt", ".ini", ".cfg"}
MAX_SNIPPET_BYTES = 12000  # 单文件摘要最多收录的字节

def sha1_bytes(b: bytes) -> str:
    h = hashlib.sha1(); h.update(b); return h.hexdigest()

def is_text_file(p: Path) -> bool:
    return p.suffix.lower() in TEXT_EXTS

def walk_files(root: Path):
    for dirpath, dirnames, filenames in os.walk(root):
        # 过滤忽略目录
        dirnames[:] = [d for d in dirnames if d not in IGNORE_DIRS and not d.startswith(".pytest_cache")]
        for fn in filenames:
            yield Path(dirpath) / fn

def build_tree(root: Path) -> str:
    """简易目录树"""
    lines = []
    def _recurse(path: Path, prefix=""):
        entries = []
        for p in sorted(path.iterdir(), key=lambda x: (x.is_file(), x.name.lower())):
            if p.name in IGNORE_DIRS: continue
            if p.is_dir():
                entries.append((p, True))
            else:
                entries.append((p, False))
        for i, (p, is_dir) in enumerate(entries):
            last = (i == len(entries) - 1)
            branch = "└─ " if last else "├─ "
            lines.append(prefix + branch + p.name + ("/" if is_dir else ""))
            if is_dir:
                ext = "   " if last else "│  "
                _recurse(p, prefix + ext)
    lines.append(root.name + "/")
    _recurse(root, "")
    return "\n".join(lines)

def summarize_python(p: Path) -> dict:
    """提取类/函数签名"""
    try:
        src = p.read_text("utf-8", errors="ignore")
        tree = ast.parse(src)
    except Exception:
        return {"classes": [], "functions": []}
    classes, funcs = [], []
    for node in tree.body:
        if isinstance(node, ast.ClassDef):
            methods = [n.name for n in node.body if isinstance(n, ast.FunctionDef)]
            classes.append({"name": node.name, "methods": methods})
        elif isinstance(node, ast.FunctionDef):
            funcs.append(node.name)
    return {"classes": classes, "functions": funcs}

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--root", default=".", help="项目根目录")
    ap.add_argument("--out", default="repo_snapshot", help="输出目录名")
    args = ap.parse_args()

    root = Path(args.root).resolve()
    out  = Path(args.out).resolve()
    out.mkdir(parents=True, exist_ok=True)
    (out / "SNIPPETS").mkdir(exist_ok=True)

    # 目录树
    (out / "TREE.txt").write_text(build_tree(root), "utf-8")

    # 索引/摘要
    index_rows = []
    arch_sections = []
    pkg_guess = []
    for f in walk_files(root):
        rel = f.relative_to(root)
        try:
            data = f.read_bytes()
        except Exception:
            continue
        sha = sha1_bytes(data)
        line_cnt = 0
        try:
            line_cnt = len(data.decode("utf-8", errors="ignore").splitlines())
        except Exception:
            pass

        index_rows.append([str(rel), f.suffix, len(data), sha, line_cnt])

        # Python 源码摘要
        if f.suffix == ".py":
            info = summarize_python(f)
            snippet = data[:MAX_SNIPPET_BYTES].decode("utf-8", errors="ignore")
            (out / "SNIPPETS" / (rel.as_posix().replace("/", "__") + ".txt")).write_text(
                f"# {rel}\n\n## Classes & Methods\n{json.dumps(info, ensure_ascii=False, indent=2)}\n\n"
                f"## Head Snippet\n```\n{snippet}\n```\n", "utf-8"
            )
            pkg_guess.append(rel.as_posix())
    # 写索引
    (out / "PY_INDEX.csv").write_text(
        "path,ext,size,sha1,lines\n" + "\n".join(",".join(map(str,r)) for r in index_rows),
        "utf-8"
    )

    # 架构文档
    architecture = textwrap.dedent(f"""
    # 项目架构快照（自动生成）

    - 生成时间：本地运行
    - 根目录：`{root}`
    - 忽略目录：{", ".join(sorted(IGNORE_DIRS))}
    - 关键 Python 文件数量：{sum(1 for r in index_rows if r[1]==".py")}

    ## 目录树
    见 `TREE.txt`

    ## 模块概览（按路径粗分）
    - **checker_ui/core/**：排程、导出逻辑（如 `tickets.py`）
    - **checker_ui/ui/**：界面层（窗口、对话框、流程图视图等）
    - **checker_ui/models/**：状态/数据模型（如 `state.py`）
    - **infra/**：线程、IO、工具
    - **.github/workflows/**：CI/构建脚本
    - ***.spec**：PyInstaller 打包规格

    ## 入口建议
    - `checker_ui/main.py`（若存在）作为图形入口
    - CI 入口：`.github/workflows/*.yml`

    ## Python 文件摘要样例
    详见 `SNIPPETS/`，每个文件包含：
    - 类/函数签名（静态解析）
    - 头部片段（便于快速定位风格与依赖）
    """).strip()
    (out / "ARCHITECTURE.md").write_text(architecture, "utf-8")

    # 打包 zip
    zip_path = out.with_suffix(".zip")
    with zipfile.ZipFile(zip_path, "w", compression=zipfile.ZIP_DEFLATED) as z:
        for p in out.rglob("*"):
            z.write(p, p.relative_to(out.parent))
    print(f"✅ 已生成：{out}/ 与 {zip_path}")
    print("把 repo_snapshot.zip 直接发到这里，我即可全面检视并给出精准补丁与回退点。")

if __name__ == "__main__":
    main()
